# Portfolio Website Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a bilingual (EN default / SR) bold single-scroll personal-brand portfolio for a .NET/Azure software engineer, statically generated and deployed on Netlify.

**Architecture:** Astro static site renders all sections to HTML; interactive/animated regions are isolated React islands using GSAP + ScrollTrigger. English is served at the root (`/`), Serbian under `/sr/`. Project content lives in Astro Content Collections (Markdown), UI strings in JSON dictionaries. Styling uses Tailwind CSS v4 with design tokens.

**Tech Stack:** Astro 5, React (islands), Tailwind CSS v4, GSAP + @gsap/react, Vitest, @fontsource (Space Grotesk + Inter), Netlify.

import en from './en.json';
import sr from './sr.json';

export const languages = { en: 'English', sr: 'Srpski' } as const;
export const defaultLang = 'en';
export type Lang = keyof typeof languages;

const dictionaries: Record<Lang, Record<string, string>> = { en, sr };

export function getLangFromUrl(url: URL): Lang {
  const [, seg] = url.pathname.split('/');
  if (seg === 'sr') return 'sr';
  return defaultLang;
}

export function useTranslations(lang: Lang) {
  return function t(key: string): string {
    return dictionaries[lang][key] ?? key;
  };
}
```

import type { Lang } from '../i18n/ui';

export interface ProjectData {
  title: string;
  description: string;
  tags: string[];
  github?: string;
  live?: string;
  order: number;
  featured: boolean;
  lang: Lang;
  slug: string;
}

export function filterByLang<T extends { lang: Lang }>(items: T[], lang: Lang): T[] {
  return items.filter((item) => item.lang === lang);
}

export function sortProjects<T extends { order: number; featured: boolean }>(items: T[]): T[] {
  return [...items].sort((a, b) => {
    if (a.featured !== b.featured) return a.featured ? -1 : 1;
    return a.order - b.order;
  });
}
```

```tsx
import { useRef, type ReactNode } from 'react';
import { gsap } from 'gsap';
import { ScrollTrigger } from 'gsap/ScrollTrigger';
import { useGSAP } from '@gsap/react';

gsap.registerPlugin(ScrollTrigger);

interface Props {
  children: ReactNode;
}

export default function Reveal({ children }: Props) {
  const el = useRef<HTMLDivElement>(null);

  useGSAP(
    () => {
      const reduce = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
      if (reduce) return;
      gsap.from(el.current, {
        y: 60,
        opacity: 0,
        duration: 0.9,
        ease: 'power3.out',
        scrollTrigger: {
          trigger: el.current,
          start: 'top 80%',
        },
      });
    },
    { scope: el }
  );

  return <div ref={el}>{children}</div>;
}
```
